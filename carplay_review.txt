import json
import random
from datetime import datetime
from pathlib import Path
from typing import Any
from zoneinfo import ZoneInfo

import yaml

PROJECT_ROOT = Path(__file__).resolve().parents[2]
STYLE_PATH = PROJECT_ROOT / "core" / "briefing" / "briefing_style.yaml"
HISTORY_PATH = PROJECT_ROOT / "data" / "briefing_history.json"


def load_style_config() -> dict[str, Any]:
    if not STYLE_PATH.exists():
        raise FileNotFoundError(f"Briefing style file not found: {STYLE_PATH}")

    with STYLE_PATH.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f) or {}


def get_now() -> datetime:
    return datetime.now(ZoneInfo("America/Los_Angeles"))


def get_time_block(now: datetime | None = None) -> str:
    now = now or get_now()
    hour = now.hour

    if 5 <= hour < 12:
        return "morning"
    if 12 <= hour < 18:
        return "afternoon"
    return "night"


def load_history() -> dict[str, Any]:
    if not HISTORY_PATH.exists():
        return {}

    try:
        return json.loads(HISTORY_PATH.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        return {}


def save_history(history: dict[str, Any]) -> None:
    HISTORY_PATH.parent.mkdir(parents=True, exist_ok=True)
    HISTORY_PATH.write_text(
        json.dumps(history, indent=2, ensure_ascii=False),
        encoding="utf-8",
    )


def choose_rotating(
    options: list[str],
    key: str,
    avoid_last: bool = True,
) -> str:
    """
    Pick a line while trying not to repeat the last used option.
    """
    if not options:
        return ""

    history = load_history()
    last_used = history.get(key)

    choices = options[:]

    if avoid_last and len(choices) > 1 and last_used in choices:
        choices.remove(last_used)

    selected = random.choice(choices)

    history[key] = selected
    save_history(history)

    return selected


def build_personality_lines(
    calendar_context: dict[str, Any],
    traffic_included: bool,
) -> dict[str, str]:
    """
    Build opener, duty/event sentence, traffic sentence, inspiration, weekday line, and closer.
    """
    style = load_style_config()
    now = get_now()
    time_block = get_time_block(now)
    weekday = calendar_context.get("weekday") or now.strftime("%A")

    openers = style.get("openers", {}).get(time_block, [])
    opener = choose_rotating(openers, f"opener_{time_block}")

    event_phrasing = style.get("event_phrasing", {})
    status = calendar_context.get("status", "unknown")
    event_count = calendar_context.get("event_count", 0)
    first_event = calendar_context.get("first_event")

    if status == "on_duty":
        duty_sentence = choose_rotating(event_phrasing.get("on_duty", []), "duty_on")
    elif status == "off_duty":
        duty_sentence = choose_rotating(event_phrasing.get("off_duty", []), "duty_off")
    else:
        duty_sentence = f"Today is marked as {calendar_context.get('status_label', status)}."

    if event_count and first_event:
        event_templates = event_phrasing.get("events_found", [])
        event_template = choose_rotating(event_templates, "events_found")
        event_sentence = event_template.format(
            count=event_count,
            first_event=first_event.get("title", "your first event"),
        )
    else:
        event_sentence = choose_rotating(event_phrasing.get("no_events", []), "no_events")

    reminders = calendar_context.get("reminders", [])
    reminder_sentence = ""
    if reminders:
        reminder_sentence = f"Reminder: {reminders[0]}"

    traffic_options = style.get("traffic_phrasing", {})
    if traffic_included:
        traffic_sentence = choose_rotating(traffic_options.get("included", []), "traffic_included")
    else:
        traffic_sentence = choose_rotating(traffic_options.get("skipped", []), "traffic_skipped")

    weekday_options = style.get("weekday_lines", {}).get(weekday, [])
    weekday_sentence = choose_rotating(weekday_options, f"weekday_{weekday}")

    inspiration_options = style.get("inspiration_lines", [])
    inspiration_sentence = choose_rotating(inspiration_options, "inspiration")

    closers = style.get("closers", [])
    closer = choose_rotating(closers, "closer")

    return {
        "opener": opener,
        "duty_sentence": duty_sentence,
        "event_sentence": event_sentence,
        "reminder_sentence": reminder_sentence,
        "traffic_sentence": traffic_sentence,
        "weekday_sentence": weekday_sentence,
        "inspiration_sentence": inspiration_sentence,
        "closer": closer,
    }


def trim_words(text: str, max_words: int = 85) -> str:
    words = text.split()

    if len(words) <= max_words:
        return text

    return " ".join(words[:max_words]).rstrip(".,;") + "."

def clean_traffic_summary(summary: str) -> str:
    """
    Keep traffic useful. If WSDOT only returns unclear/unknown sensor data,
    do not speak ugly sensor wording.
    """
    cleaned = summary.strip()
    lower = cleaned.lower()

    unclear_markers = [
        "unclear traffic",
        "unknown traffic",
        "unknown",
        "no matching",
        "unavailable",
        "sensor points",
    ]

    if any(marker in lower for marker in unclear_markers):
        return "No traffic news right now."

    return cleaned


def format_number(value: Any, fallback: str = "unknown") -> str:
    if value is None:
        return fallback

    try:
        return str(round(float(value)))
    except (TypeError, ValueError):
        return str(value)


def build_weather_sentence(weather_result: dict[str, Any]) -> str:
    """
    Build a more natural, rotating weather sentence.
    """
    style = load_style_config()
    weather_templates = style.get("weather_phrasing", [])

    forecast = weather_result.get("forecast", {}) or {}
    current = forecast.get("current", {}) or {}
    daily = forecast.get("daily", {}) or {}

    temp = format_number(current.get("temperature_2m"))
    wind = format_number(current.get("wind_speed_10m"))

    weather_code = current.get("weather_code")
    condition = weather_code_text(weather_code) if weather_code is not None else "unclear conditions"

    highs = daily.get("temperature_2m_max") or []
    lows = daily.get("temperature_2m_min") or []

    high = format_number(highs[0] if highs else None)
    low = format_number(lows[0] if lows else None)

    if weather_templates:
        template = choose_rotating(weather_templates, "weather_template")
        return template.format(
            temp=temp,
            condition=condition,
            high=high,
            low=low,
            wind=wind,
        )

    return weather_result.get("summary", "Weather check is unavailable right now.")


def build_drive_briefing(
    include_traffic_force: bool = False,
    calendar_date: date | None = None,
) -> dict[str, Any]:
    """
    Build a personalized ZÃ†NIA drive briefing using weather, traffic,
    local calendar context, and rotating personality lines.
    """
    location = select_weather_location()
    weather_result = get_weather_for_location(location)
    weather_sentence = build_weather_sentence(weather_result)

    calendar_context = build_calendar_context(calendar_date)

    traffic_config = load_traffic_config()
    active_traffic_window = get_active_window(traffic_config)

    traffic_included = False
    traffic_summary = None
    traffic_result = None

    if active_traffic_window or include_traffic_force:
        traffic_result = get_i5_traffic_summary()
        traffic_summary = clean_traffic_summary(traffic_result["summary"])
        traffic_included = True

    personality = build_personality_lines(
        calendar_context=calendar_context,
        traffic_included=traffic_included,
    )

    parts = [
        personality["opener"],
        weather_sentence,
        personality["duty_sentence"],
        personality["event_sentence"],
    ]

    if personality.get("reminder_sentence"):
        parts.append(personality["reminder_sentence"])

    if traffic_summary:
        parts.append(traffic_summary)
    else:
        parts.append(personality["traffic_sentence"])

    parts.extend(
        [
            personality["weekday_sentence"],
            personality["inspiration_sentence"],
            personality["closer"],
        ]
    )

    spoken_briefing = " ".join(part for part in parts if part).strip()
    spoken_briefing = trim_words(spoken_briefing, max_words=95)

    return {
        "weather_location": location,
        "weather_result": weather_result,
        "weather_summary": weather_sentence,
        "calendar_context": calendar_context,
        "traffic_included": traffic_included,
        "traffic_window": active_traffic_window,
        "traffic_summary": traffic_summary,
        "traffic_result": traffic_result,
        "personality": personality,
        "spoken_briefing": spoken_briefing,
    }

def load_calendar_config() -> dict[str, Any]:
    if not CALENDAR_PATH.exists():
        raise FileNotFoundError(f"ZÃ†NIA calendar file not found: {CALENDAR_PATH}")

    with CALENDAR_PATH.open("r", encoding="utf-8") as f:

def get_calendar_timezone(config: dict[str, Any]) -> ZoneInfo:
    calendar_config = config.get("calendar_context", {})
    timezone_name = calendar_config.get("timezone", "America/Los_Angeles")
    return ZoneInfo(timezone_name)


def get_today(config: dict[str, Any] | None = None) -> date:
    config = config or load_calendar_config()
    tz = get_calendar_timezone(config)
    return datetime.now(tz).date()


def get_weekday_name(target_date: date) -> str:
    return target_date.strftime("%A")


def get_status_for_date(
    config: dict[str, Any],
    target_date: date,
) -> dict[str, Any]:
    """
    Check special_dates first. If no override exists, use the default weekly schedule.
    """
    target_key = target_date.isoformat()
    weekday_name = get_weekday_name(target_date)

    special_dates = config.get("special_dates", {}) or {}
    default_schedule = config.get("default_schedule", {}) or {}
    status_phrases = config.get("status_phrases", {}) or {}

    if target_key in special_dates:
        special = special_dates[target_key] or {}
        status = special.get("status", "unknown")
        note = special.get("note")
    else:
        status = default_schedule.get(weekday_name, "unknown")
        note = None

    return {
        "status": status,
        "status_label": status_phrases.get(status, status.replace("_", " ")),
        "note": note,
        "weekday": weekday_name,
        "date": target_key,
    }


def normalize_event(event: dict[str, Any], source: str) -> dict[str, Any]:
    return {
        "date": event.get("date"),
        "day": event.get("day"),
        "time": event.get("time", ""),
        "title": event.get("title", "Untitled event"),
        "reminder": event.get("reminder"),
        "source": source,
    }


def get_events_for_date(
    config: dict[str, Any],
    target_date: date,
) -> list[dict[str, Any]]:
    """
    Return one-time and weekly recurring events that apply to the selected date.
    """
    target_key = target_date.isoformat()
    weekday_name = get_weekday_name(target_date)

    matched_events: list[dict[str, Any]] = []

    for event in config.get("events", []) or []:
        if event.get("date") == target_key:
            matched_events.append(normalize_event(event, "one_time"))

    for event in config.get("recurring_events", []) or []:
        if event.get("day") == weekday_name:
            matched_events.append(normalize_event(event, "recurring"))

    matched_events.sort(key=lambda item: item.get("time") or "99:99")
    return matched_events


def build_calendar_context(target_date: date | None = None) -> dict[str, Any]:
    """
    Build the calendar context ZÃ†NIA can use inside spoken briefings.
    """
    config = load_calendar_config()

    calendar_config = config.get("calendar_context", {})
    enabled = bool(calendar_config.get("enabled", True))

    target_date = target_date or get_today(config)
    status = get_status_for_date(config, target_date)
    events = get_events_for_date(config, target_date)

    first_event = events[0] if events else None
    reminders = [
        event["reminder"]
        for event in events
        if event.get("reminder")
    ]

    return {
        "enabled": enabled,
        "date": target_date.isoformat(),
        "weekday": get_weekday_name(target_date),
        "status": status["status"],
        "status_label": status["status_label"],
        "status_note": status["note"],
        "events": events,
        "event_count": len(events),
        "first_event": first_event,
        "reminders": reminders,
    }


def build_simple_calendar_sentence(context: dict[str, Any]) -> str:
    """
    Simple fallback sentence before the full personality layer is connected.
    """
    if not context.get("enabled", True):
        return "Calendar context is disabled."

    status_label = context.get("status_label", "unknown")
    event_count = context.get("event_count", 0)
    first_event = context.get("first_event")

    parts = [f"Today is marked as {status_label}."]

    if event_count and first_event:
        parts.append(f"You have {event_count} calendar item today. First up: {first_event['title']}.")
    else:
        parts.append("No local calendar events are listed for today.")

    reminders = context.get("reminders", [])
    if reminders:
        parts.append(f"Reminder: {reminders[0]}")

    return " ".join(parts)

VALID_STATUSES = {
    "on_duty",
    "off_duty",

def load_config() -> dict[str, Any]:
    if not CALENDAR_PATH.exists():

def save_config(config: dict[str, Any]) -> None:
    CALENDAR_PATH.parent.mkdir(parents=True, exist_ok=True)

    with CALENDAR_PATH.open("w", encoding="utf-8") as f:
        yaml.safe_dump(
            config,
            f,
            sort_keys=False,
            allow_unicode=True,
            width=1000,
        )


def validate_date(value: str) -> str:
    date.fromisoformat(value)
    return value


def validate_status(status: str) -> str:

from pathlib import Path
from typing import Any

import yaml


DAYS = [
    "Monday",
    "Tuesday",
    "Wednesday",
    "Thursday",
    "Friday",
    "Saturday",
    "Sunday",
]

TIME_BLOCKS = ["Morning", "Afternoon", "Night"]


def load_message_config(config_path: Path) -> dict[str, Any]:
    if not config_path.exists():
        raise FileNotFoundError(f"Message config not found: {config_path}")

    with config_path.open("r", encoding="utf-8") as file:
        data = yaml.safe_load(file)

    if not isinstance(data, dict):
        raise ValueError("messages.yaml is empty or invalid.")

    return data


def validate_messages(data: dict[str, Any]) -> list[str]:
    problems: list[str] = []

    messages = data.get("messages", {})
    avoid_phrases = [
        phrase.lower()
        for phrase in data.get("profile", {}).get("avoid_phrases", [])
    ]

    for day in DAYS:
        if day not in messages:
            problems.append(f"Missing day: {day}")
            continue

        for block in TIME_BLOCKS:
            message = messages[day].get(block)

            if not message:
                problems.append(f"Missing message: {day} {block}")
                continue

            lower_message = message.lower()

            for phrase in avoid_phrases:
                if phrase in lower_message:
                    problems.append(
                        f"Avoided phrase found in {day} {block}: {phrase}"
                    )

            if "=" in message:
                problems.append(f"Message contains '=' which breaks shortcut: {day} {block}")

    return problems


def build_carstartup_text(data: dict[str, Any]) -> str:
    messages = data["messages"]
    lines: list[str] = []

    for day in DAYS:
        for block in TIME_BLOCKS:
            key = f"{day}{block}"
            message = messages[day][block].strip()
            lines.append(f"{key}={message}")

    return "\n".join(lines) + "\n"


def write_carstartup_file(output_path: Path, text: str) -> None:
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(text, encoding="utf-8")


def get_default_onedrive_path() -> Path:
    home = Path.home()

    candidates = [
        home / "OneDrive" / "CarStartup.txt",
        home / "OneDrive - Personal" / "CarStartup.txt",
        home / "OneDrive" / "Documents" / "CarStartup.txt",
    ]

    for path in candidates:
        if path.parent.exists():
            return path

    return home / "OneDrive" / "CarStartup.txt"

def generate_elevenlabs_audio(text: str, output_path: Path) -> Path:
    """
    Generate an MP3 using Zack's configured ElevenLabs ZÃ†NIA voice.
    This does not play the audio. It only saves it for the iPhone Shortcut.
    """
    config = load_voice_config()

    elevenlabs_config = config.get("elevenlabs", {})
    api_key_env = elevenlabs_config.get("api_key_env", "ELEVENLABS_API_KEY")

        )

        raise RuntimeError("Missing ElevenLabs voice_id in voice_config.yaml")

    url = f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}"

    payload = {
        "text": text,

            "stability": 0.58,

            "style": 0.18,
            "use_speaker_boost": True,

    }

    headers = {
        "xi-api-key": api_key,

from datetime import date, datetime, time, timedelta
from pathlib import Path

def as_local_datetime(value: Any) -> datetime:
    """
    Convert iCal date/dateTime values into America/Los_Angeles datetime.
    """
    if isinstance(value, datetime):
        if value.tzinfo is None:
            return value.replace(tzinfo=LOCAL_TIMEZONE)
        return value.astimezone(LOCAL_TIMEZONE)

    if isinstance(value, date):
        return datetime.combine(value, time.min, tzinfo=LOCAL_TIMEZONE)

    raise TypeError(f"Unsupported calendar time value: {value!r}")


def clean_text(value: Any, max_length: int = 180) -> str:
    if value is None:
        return ""

    text = str(value).replace("\\n", " ").replace("\n", " ").strip()
    text = " ".join(text.split())

    if len(text) > max_length:
        return text[: max_length - 3].rstrip() + "..."

    return text


def build_reminder_from_event(event: Any) -> str | None:
    """
    Build a short reminder from location or description.
    """
    location = clean_text(event.get("LOCATION"), max_length=90)
    description = clean_text(event.get("DESCRIPTION"), max_length=140)

    if location and description:
        return f"Location: {location}. {description}"

    if location:
        return f"Location: {location}."

    if description:
        return description

    return None


def convert_ical_event(event: Any) -> dict[str, Any] | None:
    """
    Convert one Google iCal event into ZÃ†NIA local calendar event format.
    """
    summary = clean_text(event.get("SUMMARY"), max_length=100)

    if not summary:
        summary = "Untitled calendar event"

    dtstart_raw = event.get("DTSTART")
    if not dtstart_raw:
        return None

    start_value = dtstart_raw.dt
    start_dt = as_local_datetime(start_value)

    is_all_day = isinstance(start_value, date) and not isinstance(start_value, datetime)

    event_time = "00:00" if is_all_day else start_dt.strftime("%H:%M")
    title = f"All day: {summary}" if is_all_day else summary

    uid = clean_text(event.get("UID"), max_length=200)

    return {
        "date": start_dt.date().isoformat(),
        "time": event_time,
        "title": title,
        "reminder": build_reminder_from_event(event),
        "source": GOOGLE_SYNC_SOURCE,
        "google_uid": uid,
    }


def get_google_events_between(start_date: date, days: int = 2) -> list[dict[str, Any]]:
    """
    Pull Google Calendar events from the iCal feed.
    """
    calendar = fetch_ics_calendar()

    start_dt = datetime.combine(start_date, time.min, tzinfo=LOCAL_TIMEZONE)
    end_dt = start_dt + timedelta(days=days)

    raw_events = recurring_ical_events.of(calendar).between(start_dt, end_dt)

    converted_events: list[dict[str, Any]] = []

    for raw_event in raw_events:
        converted = convert_ical_event(raw_event)
        if converted:
            converted_events.append(converted)

    converted_events.sort(key=lambda item: (item["date"], item["time"], item["title"]))
    return converted_events


def remove_existing_google_events(
    config: dict[str, Any],
    target_dates: set[str],
) -> dict[str, Any]:
    """
    Remove old Google-synced events only for the dates being refreshed.
    Keep Zack's manually added local events.
    """
    existing_events = config.get("events") or []
    kept_events = []

    for event in existing_events:
        event_date = event.get("date")
        event_source = event.get("source")

        if event_date in target_dates and event_source == GOOGLE_SYNC_SOURCE:
            continue

        kept_events.append(event)

    config["events"] = kept_events
    return config


def sync_google_calendar_to_local(
    start_date: date | None = None,
    days: int = 2,
) -> dict[str, Any]:
    """
    Sync Google Calendar iCal events into ZÃ†NIA's local calendar file.

    By default, this imports today and tomorrow.
    """
    start_date = start_date or datetime.now(LOCAL_TIMEZONE).date()

    google_events = get_google_events_between(start_date=start_date, days=days)

    target_dates = {
        (start_date + timedelta(days=offset)).isoformat()
        for offset in range(days)
    }

    config = load_local_calendar_config()
    config = remove_existing_google_events(config, target_dates)

    local_events = config.get("events") or []
    local_events.extend(google_events)
    local_events.sort(key=lambda item: (item.get("date", ""), item.get("time", ""), item.get("title", "")))

    config["events"] = local_events
    save_local_calendar_config(config)

    return {
        "start_date": start_date.isoformat(),
        "days": days,
        "target_dates": sorted(target_dates),
        "imported_count": len(google_events),
        "events": google_events,
    }

I5_AREA_KEYWORDS = [
    "i-5",
    "i 5",
    "interstate 5",
    "lacey",
    "jblm",
    "joint base lewis",
    "mcchord",
    "dupont",

def get_wsdot_api_key() -> str:
    api_key = os.getenv("WSDOT_API_KEY")

    if not api_key:
        raise RuntimeError(
            "Missing WSDOT_API_KEY. Save it to Windows before using traffic tools."
        )

    return api_key


def fetch_traffic_flows() -> list[dict[str, Any]]:
    api_key = get_wsdot_api_key()

    response = requests.get(
        TRAFFIC_FLOWS_URL,
        params={"AccessCode": api_key},
        timeout=20,
    )

    response.raise_for_status()

    data = response.json()

    if not isinstance(data, list):
        raise RuntimeError("Unexpected WSDOT traffic flow response format.")

    return data


def item_to_search_text(item: dict[str, Any]) -> str:
    parts = []

    for key, value in item.items():
        if isinstance(value, (str, int, float)):
            parts.append(str(value))

        if isinstance(value, dict):
            for inner_value in value.values():
                if isinstance(inner_value, (str, int, float)):
                    parts.append(str(inner_value))

    return " ".join(parts).lower()


def filter_i5_area_flows(flows: list[dict[str, Any]]) -> list[dict[str, Any]]:
    matches = []

    for item in flows:
        search_text = item_to_search_text(item)

        if any(keyword in search_text for keyword in I5_AREA_KEYWORDS):
            matches.append(item)

    return matches


def summarize_flow_conditions(flows: list[dict[str, Any]]) -> str:
    if not flows:
        return "No matching I-5 traffic flow sensors were found for the JBLM, Lacey, or Tacoma area."

    condition_counts: dict[str, int] = {}

    for item in flows:
        condition = (
            item.get("CurrentFlow")
            or item.get("Flow")
            or item.get("Status")
            or item.get("TrafficStatus")
            or "Unknown"
        )

        condition = str(condition)
        condition_counts[condition] = condition_counts.get(condition, 0) + 1

    worst_order = ["StopAndGo", "Heavy", "Moderate", "WideOpen", "Unknown"]

    worst_condition = "Unknown"

    for condition in worst_order:
        if condition in condition_counts:
            worst_condition = condition
            break

    readable = {
        "WideOpen": "clear",
        "Moderate": "moderate",
        "Heavy": "heavy",
        "StopAndGo": "stop-and-go",
        "Unknown": "unclear",
    }.get(worst_condition, worst_condition)

    return (
        f"I-5 traffic check: {readable} traffic reported across "
        f"{len(flows)} matching sensor points near the JBLM, Lacey, and Tacoma corridor."
    )


def get_i5_traffic_summary() -> dict[str, Any]:
    all_flows = fetch_traffic_flows()
    i5_flows = filter_i5_area_flows(all_flows)
    summary = summarize_flow_conditions(i5_flows)

    return {
        "total_flow_points": len(all_flows),
        "matched_i5_points": len(i5_flows),
        "summary": summary,
        "matched_flows": i5_flows,
    }

def get_active_window(config: dict[str, Any]) -> str | None:
    now = get_now(config)
    current_day = now.strftime("%A")
    current_time = now.time()

    windows = config.get("check_windows", {})

    for window_name, window in windows.items():
        if not window.get("enabled", False):
            continue

        if current_day not in window.get("days", []):
            continue

        start = parse_clock(window["start_time"])
        end = parse_clock(window["end_time"])

        if start <= current_time <= end:
            return window_name

    return None

import requests


GEOCODE_URL = "https://geocoding-api.open-meteo.com/v1/search"
FORECAST_URL = "https://api.open-meteo.com/v1/forecast"


WEATHER_CODES = {
    0: "clear",
    1: "mostly clear",
    2: "partly cloudy",
    3: "cloudy",
    45: "foggy",
    48: "foggy with possible icy fog",
    51: "light drizzle",
    53: "drizzle",
    55: "heavy drizzle",
    61: "light rain",
    63: "rain",
    65: "heavy rain",
    71: "light snow",
    73: "snow",
    75: "heavy snow",
    80: "light showers",
    81: "showers",
    82: "heavy showers",
    95: "thunderstorms",
}


def weather_code_text(code: int | None) -> str:
    if code is None:
        return "unknown conditions"

    return WEATHER_CODES.get(code, f"weather code {code}")


def geocode_location(lookup_name: str) -> dict:
    params = {
        "name": lookup_name,
        "count": 1,
        "language": "en",
        "format": "json",
    }

    response = requests.get(GEOCODE_URL, params=params, timeout=15)
    response.raise_for_status()

    data = response.json()
    results = data.get("results", [])

    if not results:
        raise RuntimeError(f"No geocoding result found for: {lookup_name}")

    result = results[0]

    return {
        "name": result.get("name"),
        "latitude": result["latitude"],
        "longitude": result["longitude"],
        "timezone": result.get("timezone", "America/Los_Angeles"),
        "admin1": result.get("admin1"),
        "country": result.get("country"),
    }


def get_forecast(latitude: float, longitude: float, timezone: str = "America/Los_Angeles") -> dict:
    params = {
        "latitude": latitude,
        "longitude": longitude,
        "current": "temperature_2m,apparent_temperature,precipitation,rain,weather_code,wind_speed_10m,wind_gusts_10m",
        "daily": "temperature_2m_max,temperature_2m_min,precipitation_sum,wind_gusts_10m_max,weather_code",
        "temperature_unit": "fahrenheit",
        "wind_speed_unit": "mph",
        "precipitation_unit": "inch",
        "timezone": timezone,
        "forecast_days": 1,
    }

    response = requests.get(FORECAST_URL, params=params, timeout=15)
    response.raise_for_status()

    return response.json()


def build_weather_summary(forecast: dict, display_label: str) -> str:
    current = forecast.get("current", {})
    daily = forecast.get("daily", {})

    temp = current.get("temperature_2m")
    feels = current.get("apparent_temperature")
    condition = weather_code_text(current.get("weather_code"))
    wind = current.get("wind_speed_10m")
    gusts = current.get("wind_gusts_10m")
    rain_now = current.get("rain") or 0
    precip_today = (daily.get("precipitation_sum") or [0])[0]
    high = (daily.get("temperature_2m_max") or [None])[0]
    low = (daily.get("temperature_2m_min") or [None])[0]

    alerts = []

    if rain_now > 0 or precip_today >= 0.03 or "rain" in condition or "showers" in condition:
        alerts.append("roads may be wet")

    if gusts is not None and gusts >= 25:
        alerts.append("watch the wind gusts")

    if temp is not None and temp <= 35:
        alerts.append("watch for cold spots")

    if "fog" in condition:
        alerts.append("visibility may be lower")

    if temp is None:
        return f"Weather {display_label}: weather data is unavailable right now."

    summary = f"Weather {display_label}: {round(temp)} degrees, {condition}"

    if feels is not None and abs(feels - temp) >= 5:
        summary += f", feels like {round(feels)}"

    if high is not None and low is not None:
        summary += f". High {round(high)}, low {round(low)}"

    if wind is not None:
        summary += f". Wind around {round(wind)} miles per hour"

    if alerts:
        summary += ". " + ", ".join(alerts) + "."

    return summary


def get_weather_for_location(location: dict) -> dict:
    lookup_name = location.get("lookup_name")
    display_label = location.get("display_label", lookup_name)

    latitude = location.get("latitude")
    longitude = location.get("longitude")
    timezone = "America/Los_Angeles"

    if latitude is None or longitude is None:
        geocoded = geocode_location(lookup_name)
        latitude = geocoded["latitude"]
        longitude = geocoded["longitude"]
        timezone = geocoded.get("timezone") or timezone
    else:
        geocoded = {
            "name": location.get("name"),
            "latitude": latitude,
            "longitude": longitude,
            "timezone": timezone,
        }

    forecast = get_forecast(latitude, longitude, timezone)
    summary = build_weather_summary(forecast, display_label)

    return {
        "location": location,
        "geocoded": geocoded,
        "forecast": forecast,
        "summary": summary,
    }

def is_weekday_duty_time(config: dict[str, Any], now: datetime) -> bool:
    rule = config.get("schedule_rules", {}).get("weekday_duty_hours", {})

    if not rule.get("enabled", False):
        return False

    current_day = now.strftime("%A")
    allowed_days = rule.get("days", [])

    if current_day not in allowed_days:
        return False

    start = parse_time(rule["start_time"])
    end = parse_time(rule["end_time"])

    current_time = now.time()

    return start <= current_time <= end


def select_weather_location() -> dict[str, Any]:
    config = load_weather_config()

    manual = config.get("manual_override", {})
    locations = config.get("locations", {})

    if manual.get("enabled"):
        location_key = manual.get("location_key")

        if location_key and location_key in locations:
            selected = locations[location_key].copy()
            selected["source"] = "manual_override"
            selected["location_key"] = location_key

                "lookup_name": custom_lookup,
                "latitude": None,
                "longitude": None,
                "source": "manual_override_custom",
                "location_key": "custom",
            }

    now = get_now_for_config(config)

    if is_weekday_duty_time(config, now):
        location_key = config["schedule_rules"]["weekday_duty_hours"]["location_key"]
    else:
        location_key = config.get("default_location_key", "off_duty")

    selected = locations[location_key].copy()
    selected["source"] = "schedule"
    selected["location_key"] = location_key
    selected["selected_at"] = now.isoformat()

    return selected
